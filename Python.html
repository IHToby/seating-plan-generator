#!/usr/bin/env python3 import random def are_seats_near(row1, col1, row2, col2): """Returns True if the two seats are near (within one row and one column).""" return abs(row1 - row2) <= 1 and abs(col1 - col2) <= 1 def is_seat_valid(student, row, col, assigned_seats, restrictions): """ Check if placing the student at (row, col) does not violate any of the seating restrictions. restrictions: list of tuples [(stu1, stu2), ...] that must not sit near each other. """ for restricted1, restricted2 in restrictions: if student == restricted1 and restricted2 in assigned_seats: r, c = assigned_seats[restricted2] if are_seats_near(row, col, r, c): return False if student == restricted2 and restricted1 in assigned_seats: r, c = assigned_seats[restricted1] if are_seats_near(row, col, r, c): return False return True def assign_student(student, height, rows, cols, available_seats, assigned_seats, restrictions): """Try to assign a single student to a valid seat.""" front_half_rows = (rows + 1) // 2 # Ceil division for front half rows attempts = len(available_seats) while attempts > 0: seat = random.choice(available_seats) row, col = seat # For tall students, assign in front half; for short students in back half; medium anywhere. if ((height == 'tall' and row < front_half_rows) or (height == 'short' and row >= front_half_rows) or (height == 'medium')): if is_seat_valid(student, row, col, assigned_seats, restrictions): assigned_seats[student] = (row, col) available_seats.remove(seat) return True attempts -= 1 return False def assign_pair(student1, student2, height1, height2, rows, cols, available_seats, assigned_seats, restrictions): """Try to assign a pair of students into adjacent seats.""" front_half_rows = (rows + 1) // 2 attempts = len(available_seats) while attempts > 0: seat = random.choice(available_seats) row, col = seat # Check if one of the pair is tall or short to determine row preference if (((height1 == 'tall' or height2 == 'tall') and row < front_half_rows) or ((height1 == 'short' or height2 == 'short') and row >= front_half_rows) or (height1 == 'medium' and height2 == 'medium')): if is_seat_valid(student1, row, col, assigned_seats, restrictions): # Attempt to assign student2 in the adjacent column (either col+1 or col-1) col2 = col + 1 if col % 2 == 0 else col - 1 if 0 <= col2 < cols and (row, col2) in available_seats: if is_seat_valid(student2, row, col2, assigned_seats, restrictions): assigned_seats[student1] = (row, col) assigned_seats[student2] = (row, col2) available_seats.remove(seat) available_seats.remove((row, col2)) return True attempts -= 1 return False def generate_plan(rows, cols, total_students, tall_students, medium_students, short_students, restrictions, together): """ Generate the seating plan using the provided parameters. - rows, cols: dimensions of the seating grid - total_students: total number of students (must equal the sum of tall, medium and short) - tall_students, medium_students, short_students: lists of student identifiers - restrictions: list of pairs (tuples) that must NOT sit near each other - together: list of pairs (tuples) that should be seated side-by-side """ # Check capacity and consistency if total_students > rows * cols: print("Total number of students exceeds the classroom capacity.") return None if len(tall_students) + len(medium_students) + len(short_students) != total_students: print("Number of students in height categories does not match the total number of students.") return None # Check for contradictions between paired-up and separated restrictions for together_pair in together: together1, together2 = together_pair for r in restrictions: if (r[0] == together1 and r[1] == together2) or (r[0] == together2 and r[1] == together1): print('Input in "Separate Students" contradicts that in "Pair Up Students".') return None # Create a list of all available seats as tuples (row, col) available_seats = [(i, j) for i in range(rows) for j in range(cols)] assigned_seats = {} # Create a list of all students (with their height information) students_to_assign = ([(s, 'tall') for s in tall_students] + [(s, 'medium') for s in medium_students] + [(s, 'short') for s in short_students]) random.shuffle(students_to_assign) # Process the "pair up" assignments first for pair in together: student1, student2 = pair # Determine their height category from the provided lists height1 = 'tall' if student1 in tall_students else ('medium' if student1 in medium_students else 'short') height2 = 'tall' if student2 in tall_students else ('medium' if student2 in medium_students else 'short') if not assign_pair(student1, student2, height1, height2, rows, cols, available_seats, assigned_seats, restrictions): print(f"Unable to place the pair {student1} and {student2} with the given restrictions.") return None # Process the remaining individual assignments all_assigned = True for student, height in students_to_assign: if student not in assigned_seats: if not assign_student(student, height, rows, cols, available_seats, assigned_seats, restrictions): all_assigned = False break if not all_assigned: print("Unable to place all students with the given restrictions. Try to run the function again.") return None # Create the final seating plan as a 2D list. If no student is assigned, mark it as "Empty" plan = [["Empty" for j in range(cols)] for i in range(rows)] for student, (r, c) in assigned_seats.items(): # Identify the student's height for display purposes if student in tall_students: height_class = 'tall' elif student in medium_students: height_class = 'medium' elif student in short_students: height_class = 'short' else: height_class = '' plan[r][c] = f"{student} ({height_class})" return plan def print_plan(plan): """Print the seating plan as a table to the terminal.""" if plan is None: return row_separator = "-" * (len(plan[0]) * 15) for row in plan: print(row_separator) # Adjust the spacing as needed print(" | ".join(f"{cell:^13}" for cell in row)) print(row_separator) def parse_number_list(input_str): """ Parse an input string with numbers separated by commas into a list of integers. For example, "1,2,5" => [1, 2, 5] """ # Filter out any empty strings and non-numeric values return [int(num.strip()) for num in input_str.split(",") if num.strip().isdigit()] def parse_pairs(input_str): """ Parse an input string with pairs in the form "a-b,c-d" into a list of tuples. For example, "1-2,5-6" => [(1, 2), (5, 6)] """ pairs = [] for pair_str in input_str.split(","): parts = pair_str.strip().split("-") if len(parts) == 2 and all(p.strip().isdigit() for p in parts): pairs.append((int(parts[0].strip()), int(parts[1].strip()))) return pairs def main(): # For demonstration, you can either read input from the user or set sample values. print("Seating Plan Generator") try: rows = int(input("Enter number of rows (default 4): ") or "4") cols = int(input("Enter number of columns (default 8): ") or "8") total_students = int(input("Enter total number of students (default 32): ") or "32") # Enter student identifiers (here we assume students are identified by numbers) print("Enter student numbers for each height category as comma-separated values.") tall_input = input("Tall Students (e.g., 1,2,3,...): ") medium_input = input("Medium Height Students (e.g., 4,5,6,...): ") short_input = input("Short Students (e.g., 7,8,9,...): ") tall_students = parse_number_list(tall_input) medium_students = parse_number_list(medium_input) short_students = parse_number_list(short_input) restrictions_input = input('Separate Students (comma-separated pairs, e.g., "1-2,5-6"): ') together_input = input('Pair Up Students (comma-separated pairs, e.g., "3-4,7-8"): ') restrictions = parse_pairs(restrictions_input) together = parse_pairs(together_input) except Exception as e: print("Error in input format:", e) return # Validate that the number of students equals the sum of the individuals in each category. if len(tall_students) + len(medium_students) + len(short_students) != total_students: print("Error: The sum of tall, medium, and short students does not match the total number of students.") return plan = generate_plan(rows, cols, total_students, tall_students, medium_students, short_students, restrictions, together) if plan: print("\nGenerated Seating Plan:") print_plan(plan) else: print("Seating plan generation failed due to input constraints.") if __name__ == "__main__": main()
